function startChat() {
  const role = document.getElementById('roleSelect').value;
  const hotel = document.getElementById('hotelSelect').value;
  if (!role || !hotel) { alert('Selecteer eerst jouw functie en hotel.'); return; }

  userRole = role;
  userHotel = hotel;

  document.getElementById('onboarding').style.display = 'none';
  const chat = document.getElementById('chat');

  const bar = document.getElementById('contextBar');

function buildCategoryTabs() {
  const tabs = document.getElementById('categoryTabs');
  tabs.innerHTML = '';

  Object.keys(categories).forEach(cat => {
    const tab = document.createElement('button');
    tab.className = `cat-tab${cat === activeCategory ? ' active' : ''}`;
    tab.textContent = cat;
    tab.onclick = () => selectCategory(cat, tab);
    tabs.appendChild(tab);
  });

  buildQuickButtons(activeCategory);
}

function selectCategory(cat, tabEl) {
  activeCategory = cat;
  document.querySelectorAll('.cat-tab').forEach(t => t.classList.remove('active'));
  tabEl.classList.add('active');
  buildQuickButtons(cat);
}

function buildQuickButtons(cat) {
  const wrap = document.getElementById('quickWrap');
  wrap.innerHTML = '';
  categories[cat].forEach(q => {
    const btn = document.createElement('button');
    btn.className = 'quick-btn';
    btn.textContent = q;
    btn.onclick = () => sendText(q);
    wrap.appendChild(btn);
  });
}

function addMessage(type, text, suggestions = null) {
  const messages = document.getElementById('messages');

  const wrap = document.createElement('div');
  wrap.className = `msg ${type === 'user' ? 'user' : ''}`;

  const av = document.createElement('div');
  av.className = `avatar ${type === 'bot' ? 'bot' : 'user-av'}`;
  av.textContent = type === 'bot' ? 'TH' : '👤';

  const bubble = document.createElement('div');
  bubble.className = `bubble ${type === 'bot' ? 'bot' : 'user'}`;
  bubble.innerHTML = formatText(text);

  if (suggestions && suggestions.length > 0) {
    const cards = document.createElement('div');
    cards.className = 'suggestion-cards';
    suggestions.forEach(s => {
      const btn = document.createElement('button');
      btn.className = 'suggestion-btn';
      btn.textContent = '▶ ' + s;
      btn.onclick = () => sendText(s);
      cards.appendChild(btn);
    });
    bubble.appendChild(cards);
  }

  wrap.appendChild(av);
  wrap.appendChild(bubble);
  messages.appendChild(wrap);
  messages.scrollTop = messages.scrollHeight;
}

function formatText(text) {
  return text
    .replace(/\*\*(.*?)\*\*/g, '<strong>$1</strong>')
    .replace(/\*(.*?)\*/g, '<em>$1</em>')
    .replace(/^• (.+)$/gm, '<li>$1</li>')
    .replace(/(<li>.*<\/li>)/s, '<ul>$1</ul>')
    .replace(/\n/g, '<br>');
}

function showTyping() {
  const messages = document.getElementById('messages');
  const wrap = document.createElement('div');
  wrap.className = 'typing-wrap';
  wrap.id = 'typing';

  const av = document.createElement('div');
  av.className = 'avatar bot';
  av.textContent = 'TH';

  const dots = document.createElement('div');
  dots.className = 'typing-dots';
  dots.innerHTML = '<span></span><span></span><span></span>';

  wrap.appendChild(av);
  wrap.appendChild(dots);
  messages.appendChild(wrap);
  messages.scrollTop = messages.scrollHeight;
}

function hideTyping() {
  const t = document.getElementById('typing');
  if (t) t.remove();
}

async function sendText(text) {
  if (!text.trim()) return;

  addMessage('user', text);

  const contextPrefix = `[Functie: ${roleLabels[userRole]} | Hotel: ${hotelLabels[userHotel]}]\n\n`;
  history.push({ role: 'user', content: contextPrefix + text });

  showTyping();

  const isScenario = text.toLowerCase().includes('speel') || text.toLowerCase().includes('scenario');

  try {
    const res = await fetch('https://api.anthropic.com/v1/messages', {
      method: 'POST',
headers: {
  'Content-Type': 'application/json',
  'x-api-key': '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',
  'anthropic-version': '2023-06-01'
},    });

    const data = await res.json();
    hideTyping();

    if (data.error) {
      showError('Er is een fout opgetreden: ' + (data.error.message || 'Probeer het opnieuw.'));
      history.pop();
      return;
    }

    const reply = data.content[0].text;
    history.push({ role: 'assistant', content: reply });

    let suggestions = null;
    if (isScenario) {
      suggestions = [
        'Ga verder met dit scenario',
        'Speel een ander scenario',
        'Wat zou Townhouse verwachten in deze situatie?'
      ];
    }

    addMessage('bot', reply, suggestions);

  } catch (err) {
    hideTyping();
    showError('Geen verbinding. Controleer je internetverbinding en probeer opnieuw.');
    history.pop();
  }
}

function showError(msg) {
  const messages = document.getElementById('messages');
  const err = document.createElement('div');
  err.className = 'error-msg';
  err.textContent = msg;
  messages.appendChild(err);
  messages.scrollTop = messages.scrollHeight;
}

function sendMessage() {
  const input = document.getElementById('userInput');
  const text = input.value.trim();
  if (!text) return;
  input.value = '';
  input.style.height = 'auto';
  sendText(text);
}

function handleKey(e) {
  if (e.key === 'Enter' && !e.shiftKey) {
    e.preventDefault();
    sendMessage();
  }
}

function autoResize(el) {
  el.style.height = 'auto';
  el.style.height = Math.min(el.scrollHeight, 100) + 'px';
}

function resetChat() {

  document.getElementById('messages').innerHTML = '';
  document.getElementById('onboarding').style.display = 'flex';
  document.getElementById('chat').style.display = 'none';
  document.getElementById('roleSelect').value = '';
  document.getElementById('hotelSelect').value = '';
  userRole = '';
  userHotel = '';
}
